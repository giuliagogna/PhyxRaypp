117template<typename L, typename R, typename Res> Res _sum (const L& left, const R& right) {
118 return Res{
119 left.x + right.x,
120 left.y + right.y,
121 left.z + right.z
122 };
123}
124
125template<typename L, typename R, typename Res> Res _difference (const L& left, const R& right) {
126 return Res{
127 left.x - right.x,
128 left.y - right.y,
129 left.z - right.z
130 };
131}
132
133template<typename L, typename R, typename Res> Res _negate (const L& left) {
134 return Res{
135 -left.x,
136 -left.y,
137 -left.z
138 };
139}
140
141template<typename L, typename R, typename Res> Res _elementwise_product (const L& left, const R& right) {
142 return Res{
143 left.x * right.x,
144 left.y * right.y,
145 left.z * right.z
146 };
147}
148
149template<typename L, typename R, typename Res> Res _cross_product (const L& left, const R& right) {
150 return Res{
151 left.y * right.z - left.z * right.y,
152 left.z * right.x - left.x * right.z,
153 left.x * right.y - left.y * right.x
154 };
155}
156
157template<typename L, typename R, typename Res> Res _scalar_multiply (const L& left, const R& scalar) {
158 return Res{
159 left.x * scalar,
160 left.y * scalar,
161 left.z * scalar
162 };
163}
164
165template<typename L, typename R, typename Res> Res _scalar_divide (const L& left, const R& scalar) {
166 return Res{
167 left.x / scalar,
168 left.y / scalar,
169 left.z / scalar
170 };
171}
172
173template<typename Curr, typename Res> Res _same (const Curr& left) {
174 return Res{
175 left.x,
176 left.y,
177 left.z
178 };
179}

354 Point operator* (const HomMatrix& M, const Point& p) {
355
356 float px = p.x; float py = p.y; float pz = p.z; // This should be optimized by the compiler to avoid overhead of multiple accesses to p.x, p.y and p.z
357
358 Point res{
359 M.mat[0] * px + M.mat[1] * py + M.mat[2] * pz + M.mat[3],
360 M.mat[4] * px + M.mat[5] * py + M.mat[6] * pz + M.mat[7],
361 M.mat[8] * px + M.mat[9] * py + M.mat[10] * pz + M.mat[11]
362 };
363
364 float w = M.mat[12] * px + M.mat[13] * py + M.mat[14] * pz + M.mat[15];
365
366 if (w==1.f) {
367 return res;
368 }
369
370 return res / w; // homogeneous division
371 }

442 Transformation Trans(const Vec& v) {
443 Transformation t; // Starts as Identity
444 // Set the last column to the components of the vector
445 t.m.mat[3] = v.x;
446 t.m.mat[7] = v.y;
447 t.m.mat[11] = v.z;
448 // Inverse is just a translation by -v
449 t.invm.mat[3] = -v.x;
450 t.invm.mat[7] = -v.y;
451 t.invm.mat[11] = -v.z;
452 return t;
453 }
454
456 Transformation Scale(const Vec& v) {
457 Transformation t; // Starts as Identity
458 // Set diagonal elements to components of the scaling vector
459 t.m.mat[0] = v.x;
460 t.m.mat[5] = v.y;
461 t.m.mat[10] = v.z;
462 // Inverse is just a scaling by 1/v
463 t.invm.mat[0] = 1.0f / v.x;
464 t.invm.mat[5] = 1.0f / v.y;
465 t.invm.mat[10] = 1.0f / v.z;
466 return t;
467 }

470 // Euler angles rotations (intrinsic rotations around the axes of the reference system, applied in order Z, Y, X)
472 Transformation R_x(float angle_rad) {
473 Transformation t;
474 float c = std::cos(angle_rad);
475 float s = std::sin(angle_rad);
476 // Already has 1 as mat[0]
477 t.m.mat[5] = c; t.m.mat[6] = -s;
478 t.m.mat[9] = s; t.m.mat[10] = c;
479 // Inverse of a rotation matrix is its transpose (or a rotation by -angle)
480 t.invm.mat[5] = c; t.invm.mat[6] = s;
481 t.invm.mat[9] = -s; t.invm.mat[10] = c;
482 return t;
483 }
484
486 Transformation R_y(float angle_rad) {
487 Transformation t;
488 float c = std::cos(angle_rad);
489 float s = std::sin(angle_rad);
490 // Already has 1 in mat[5]
491 t.m.mat[0] = c; t.m.mat[2] = s;
492 t.m.mat[8] = -s; t.m.mat[10] = c;
493 // Inverse
494 t.invm.mat[0] = c; t.invm.mat[2] = -s;
495 t.invm.mat[8] = s; t.invm.mat[10] = c;
496 return t;
497 }
498
500 Transformation R_z(float angle_rad) {
501 Transformation t;
502 float c = std::cos(angle_rad);
503 float s = std::sin(angle_rad);
504 // Already has 1 in mat[10]
505 t.m.mat[0] = c; t.m.mat[1] = -s;
506 t.m.mat[4] = s; t.m.mat[5] = c;
507 // Inverse
508 t.invm.mat[0] = c; t.invm.mat[1] = s;
509 t.invm.mat[4] = -s; t.invm.mat[5] = c;
510 return t;
511 }

773

775 float b = normal.x * normal.y * a;

787 };
788

793 };
