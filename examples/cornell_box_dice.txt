# Pre-computed variables for rotations
float pi_over_2 = 1.5707963;
float minus_pi_over_2 = -1.5707963;

background(<0.5, 0.7, 1.0>);
camera(perspective, 1.0, 3.0);

plane(translation([0.0, 2.0, 0.0]) * rot_x(pi_over_2), right_wall_mat);
plane(translation([6.0, 0.0, 0.0]) * rot_y(pi_over_2), floor_mat);

plane(translation([0.0, 0.0, 2.0]), floor_mat);
# Light
cube(translation([2.0, 0.0, 1.95]) * scaling([0.8, 0.8, 0.05]), lamp_mat);
