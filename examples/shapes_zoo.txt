# Pre-computed variables for rotations
float pi_over_2 = 1.5707963;
float minus_pi_over_2 = -1.5707963;
float pi_over_3 = 1.0471975;
float minus_pi_over_3 = -1.0471975;

background(<0.5, 0.7, 1.0>);
camera(perspective, 1.333333, 3.0, rot_y(0.25) * translation([0.0, 0.0, 1.0]));

# Material Definitions
material floor_mat(diffuse(uniform(<0.95, 0.85, 0.72>)));
material left_wall_mat(diffuse(uniform(<0.9, 0.0, 0.0>)));
material right_wall_mat(diffuse(uniform(<0.0, 0.9, 0.0>)));
material dice_mat(diffuse(uniform(<0.4, 0.1, 0.7>)));
material dice_dots_mat(diffuse(uniform(<0.7, 0.7, 1.0>)));
material lamp_mat(diffuse(uniform(<1.0, 1.0, 1.0>)), uniform(<4.0, 4.0, 4.0>));
material blue_mat(diffuse(uniform(<0.4, 0.4, 1.0>)));
material mirror_mat(specular(uniform(<0.6, 0.6, 0.6>)));

# ========================================================
# CORNELL BOX
# ========================================================
plane(translation([0.0, 0.0, -2.0]) * rot_z(pi_over_2), floor_mat);
plane(translation([0.0, -3.0, 0.0]) * rot_x(pi_over_2), left_wall_mat);
plane(translation([0.0, 3.0, 0.0]) * rot_x(pi_over_2), right_wall_mat);
plane(translation([7.0, 0.0, 0.0]) * rot_y(pi_over_2), floor_mat);
plane(translation([-2.7, 0.0, 0.0]) * rot_y(pi_over_2), floor_mat);
plane(translation([0.0, 0.0, 2.2]), floor_mat);
# Light
cube(translation([1.5, 0.0, 2.15]) * scaling([0.8, 0.8, 0.05]), lamp_mat);


# ========================================================
# DICE
# ========================================================

    csg_intersection(
        csg_intersection(
            cube(identity, dice_mat),
            cylinder(scaling([1.3, 1.3, 1.3]), dice_mat)
        ),
        csg_intersection(
            cylinder(rot_y(pi_over_2) * scaling([1.3, 1.3, 1.3]), dice_mat),
            cylinder(rot_x(pi_over_2) * scaling([1.3, 1.3, 1.3]), dice_mat)
        )        
    ),
    # Dice dots, well allocated tree
    csg_union(
        csg_union(
            csg_union(
                csg_union(
                    csg_union(
                        sphere(translation([-1.03,  0.5,  0.6]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat),
                        sphere(translation([-1.03,  0.5,  0.0]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat)
                    ),
                    sphere(translation([-1.03,  0.5, -0.6]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat)

                csg_union(
                    csg_union(
                        sphere(translation([-1.03, -0.5,  0.6]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat),
                        sphere(translation([-1.03, -0.5,  0.0]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat)
                    ),
                    sphere(translation([-1.03, -0.5, -0.6]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat)
                )
            ),
            csg_union(
                csg_union(
                    csg_union(
                        sphere(translation([1.05,  0.0,  0.0]) * scaling([0.3, 0.3, 0.3]), dice_dots_mat),
                        sphere(translation([0.0,  1.03,  0.0]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat)
                    ),
                    sphere(translation([0.6,  1.03,  0.6]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat)
                ),
                csg_union(
                    csg_union(
                        sphere(translation([0.6,  1.03,  -0.6]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat),
                        sphere(translation([ 0.0,  0.0, -1.03]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat)
                    ),
                    sphere(translation([-0.6,  1.03,  0.6]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat)
                )
            )
        ),
        csg_union(
            csg_union(
                csg_union(
                    csg_union(
                        sphere(translation([-0.6,  1.03,  -0.6]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat),
                        sphere(translation([-0.5, -1.03,  0.5]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat)

                    sphere(translation([ 0.5, -1.03, -0.5]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat)

            csg_union(

                    sphere(translation([-0.6, 0.6, 1.03]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat),
                    sphere(translation([ 0.6, 0.6, 1.03]) * scaling([0.2, 0.2, 0.2]), dice_dots_mat)

    translation([4.0, -2.0, 0.11]) * scaling([0.3, 0.3, 0.3]) * rot_y(1.0) * rot_z(1.0)
