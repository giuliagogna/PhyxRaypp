115export struct FlatRenderer : Renderer {
116 // Inherit constructors from Renderer
117 using Renderer::Renderer;
118
119 [[nodiscard]] Color operator()(const Ray& ray) const override {
120 auto hit = world->ray_intersection(ray);
121
122 if (hit.has_value()) {
123 const Shape* shape = hit->hitted_shape;
124
125 // Safety check: ensure the shape has a material and a BRDF attached
126 if (shape->material && shape->material->brdf) {
127 return shape->material->brdf->pigment->get_color(hit->surface_params);
128 }
129
130 // Fallback color if the shape has no material attached
131 return Color{1.0f, 0.0f, 0.0f};
132 }
133
134 return background_color;
135 }
136};

147export struct PathTracer : Renderer {
148
150 PCG& pcg;
151
153 int num_of_rays;
154
156 int max_depth;
157
159 int russian_roulette_limit;
160
161 using Renderer::Renderer;
162
180 PathTracer(
181 PCG& pcg,
182 World* world,
183 Color background_color = Color{0.0f, 0.0f, 0.0f},
184 int num_of_rays = 10,
185 int max_depth = 10,
186 int russian_roulette_limit = 3
187 ) : Renderer(world, background_color), pcg(pcg), num_of_rays(num_of_rays), max_depth(max_depth), russian_roulette_limit(russian_roulette_limit) {}
188
189 [[nodiscard]] Color operator()(const Ray& ray) const override {
190
191 // Stop recursion once the maximum depth is reached.
192 if(ray.depth > max_depth) {
193 return Color{0.0f, 0.0f, 0.0f};
194 }
195
196 auto hit_record = world->ray_intersection(ray);
197
198 // No intersection: terminate the path and return the background color.
199 if (!hit_record.has_value()) {
200 return background_color;
201 }
202
203 // If the ray intersects something, we need to solve the rendering equation at the point of intersection
204 auto hit_material = hit_record->hitted_shape->material;
205
206 Color hit_color = hit_material->brdf->pigment->get_color(hit_record->surface_params); // Surface reflectance.
207 Color emitted_radiance = hit_material->emitted_radiance->get_color(hit_record->surface_params); // Self-emitted radiance.
208
209 // ---------------------------------------------------------------------------------------------------
210 // Russian Roulette: triggered only if the ray has been reflected at least russian_roulette_limit
211 // ---------------------------------------------------------------------------------------------------
212
213 // Estimate a continuation probability from the surface reflectance.
214 float hit_color_lum = std::max({hit_color.r, hit_color.g, hit_color.b});
215
216 if (ray.depth > russian_roulette_limit) { // First reflections are always preserved
217 float q = std::max(0.05f, 1.0f-hit_color_lum);
218 if (pcg.random_float() > q) {
219 // Continue the path and compensate for discarded samples to keep the estimator unbiased.
220 hit_color *= 1.0f/(1.0f - q);
221 } else {
222 // Terminate the path.
223 return emitted_radiance;
224 }
225 }
226
227 // Estimate indirect radiance by recursively tracing scattered rays.
228
229 // ---------------------------------------------------------------------------------------------------
230 // Monte Carlo estimate of indirect radiance.
231 // ---------------------------------------------------------------------------------------------------
232
233 Color cum_radiance = Color{0.0f, 0.0f, 0.0f};
234
235 if (hit_color_lum > 0.0f) { // Only do costly recursions if it's worth it
236 for (int ray_index = 0; ray_index < num_of_rays; ray_index++) {
237
238 // Sample a scattered direction from the BRDF.
239 Ray new_ray = hit_material->brdf->scatter_ray(
240 pcg,
241 hit_record->ray.direction,
242 hit_record->hit_point,
243 hit_record->hit_normal,
244 hit_record->ray.depth + 1
245 );
246
247 // Recursively evaluate incoming radiance.
248 Color new_radiance = (*this)(new_ray);
249 cum_radiance += hit_color * new_radiance; // Add the radiance of the new ray, obtained by recursion, to the current radiance
250 }
251 }
252
253 // Direct emission + average indirect contribution.
254 return emitted_radiance + cum_radiance * 1.0f / num_of_rays;
255 }
