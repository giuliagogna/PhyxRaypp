66// =========================================================================
67// FLAT RENDERER (Pigment colors)
68// =========================================================================
69export struct FlatRenderer : Renderer {
70 // Inherit constructors from Renderer
71 using Renderer::Renderer;
72
73 [[nodiscard]] Color operator()(const Ray& ray) const override {
74 auto hit = world->ray_intersection(ray);
75
76 if (hit.has_value()) {
77 const Shape* shape = hit->hitted_shape;
78
79 // Safety check: ensure the shape has a material and a BRDF attached
80 if (shape->material && shape->material->brdf) {
81 return shape->material->brdf->pigment->get_color(hit->surface_params);
82 }
83
84 // Fallback color (Red) if the shape has no material attached
85 return Color{1.0f, 0.0f, 0.0f};
86 }
87
88 return background_color;
89 }
90};

92// =========================================================================
93// PATH TRACER (solution of rendering equation)
94// =========================================================================
95
96export struct PathTracer : Renderer {
97
98 PCG& pcg;
99 int num_of_rays;
100 int max_depth;
101 int russian_roulette_limit;
102
103 using Renderer::Renderer;
104
105 PathTracer(
106 PCG& pcg,
107 World* world,
108 Color background_color = Color{0.0f, 0.0f, 0.0f},
109 int num_of_rays = 10,
110 int max_depth = 10,
111 int russian_roulette_limit = 3
112 ) : Renderer(world, background_color), pcg(pcg), num_of_rays(num_of_rays), max_depth(max_depth), russian_roulette_limit(russian_roulette_limit) {}
113
114 [[nodiscard]] Color operator()(const Ray& ray) const override {
115
116 // If the ray has been reflected too many times return black
117 // This happens at the end of recursion if it's not truncated by the Russian Roulette before
118 if(ray.depth > max_depth) {
119 return Color{0.0f, 0.0f, 0.0f};
120 }
121
122 // If the ray does not intersect anything return the background (emissive) color and stops the recursion
123 auto hit_record = world->ray_intersection(ray);
124 if (!hit_record.has_value()) {
125 return background_color;
126 }
127
128 // If the ray intersects something, we need to solve the rendering equation at the point of intersection
129 auto hit_material = hit_record->hitted_shape->material;
130 Color hit_color = hit_material->brdf->pigment->get_color(hit_record->surface_params); // the Color it "reflects"
131 Color emitted_radiance = hit_material->emitted_radiance->get_color(hit_record->surface_params); // the Color it emits
132
134 // Russian Roulette: triggered only if the ray has been reflected at least russian_roulette_limit
136 // The probability of terminating the recursion is based on the luminosity of the current hit
137 // "Load the bullets!"
138 float hit_color_lum = std::max({hit_color.r, hit_color.g, hit_color.b});
139 // "Pull the trigger!"
140 if (ray.depth > russian_roulette_limit) { // First reflections are mandatory
141 float q = std::max(0.05f, 1.0f-hit_color_lum);
142 if (pcg.random_float() > q) {
143 // Keep the recursion going, but compensate for other potentially discarded rays
144 hit_color *= 1.0f/(1.0f - q);
145 } else {
146 // Stop the recursion prematurely
147 return emitted_radiance;
148 }
149 }
150
151 // If the ray survived the Russian Roulette, it's time for the recursive call...
152
153 // Montecarlo integration
154 Color cum_radiance = Color{0.0f, 0.0f, 0.0f};
155
156 if (hit_color_lum > 0.0f) { // Only do costly recursions if it's worth it
157 for (int ray_index = 0; ray_index < num_of_rays; ray_index++) {
158
159 Ray new_ray = hit_material->brdf->scatter_ray(
160 pcg,
161 hit_record->ray.direction,
162 hit_record->hit_point,
163 hit_record->hit_normal,
164 hit_record->ray.depth + 1
165 );
166
167 Color new_radiance = (*this)(new_ray);
168 cum_radiance += hit_color * new_radiance; // Add the radiance of the new ray, obtained by recursion, to the current radiance
169 }
170 }
171
172 return emitted_radiance + cum_radiance * 1.0f / num_of_rays;
173 }
174};
