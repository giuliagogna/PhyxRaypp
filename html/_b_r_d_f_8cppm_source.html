45// Pure isotropic diffusion BRDF
46export struct DiffusiveBRDF : BRDF {
47 DiffusiveBRDF(std::shared_ptr<Pigment> pigment = std::make_shared<UniformPigment>(Color{1.0f, 1.0f, 1.0f})) :
48 BRDF(pigment = std::move(pigment)) {}
49
50 Ray scatter_ray(PCG& pcg, Vec incoming_direction, Point interaction_point, Normal normal, int depth) override {
51 auto [e1, e2, e3] = create_onb_from_z(normal);
52
53 float cos_theta_sq = pcg.random_float();
54 float cos_theta = std::sqrt(cos_theta_sq);
55 float sin_theta = std::sqrt(1 - cos_theta_sq);
56 float phi = 2.0f * std::numbers::pi_v<float> * pcg.random_float();
57
58 return Ray{
59 interaction_point,
60 e1*std::cos(phi)*sin_theta + e2*std::sin(phi)*sin_theta + e3*cos_theta,
61 1.0e-3f,
62 std::numeric_limits<float>::infinity(),
63 depth
64 };
65 };
66};

68// Specular reflection BRDF, chosen sharpness
69export struct SpecularBRDF : BRDF {
70 float threshold_angle_rad;
71
72 SpecularBRDF(std::shared_ptr<Pigment> pigment = std::make_shared<UniformPigment>(Color{1.0f, 1.0f, 1.0f}),
73 float threshold_angle = std::numbers::pi_v<float> / 1800.0f) :
74 BRDF(std::move(pigment)), threshold_angle_rad(threshold_angle) {}
75
76 Ray scatter_ray(PCG& pcg, Vec incoming_direction, Point interaction_point, Normal normal, int depth) override {
77 Vec ray_dir = incoming_direction.normalize();
78 Vec normal_vec = Vec{normal.x, normal.y, normal.z}.normalize();
79 float dot_product = normal_vec * ray_dir;
80
81 return Ray{
82 interaction_point,
83 ray_dir - normal_vec * 2 * dot_product,
84 1.0e-3f,
85 std::numeric_limits<float>::infinity(),
86 depth
87 };
88 };
