35 // ===================================================================
36 // Checks if two objects with three coordinates x, y and z are close
37 // ===================================================================
38 template<typename T>
39 bool are_xyz_close(const T& a, const T& b, float epsilon = 1e-5f) {
40 return aux::are_close(a.x, b.x, epsilon) &&
41 aux::are_close(a.y, b.y, epsilon) &&
42 aux::are_close(a.z, b.z, epsilon);
43 }

45 // =========================================================
46 // Managing of input/output file opening
47 // =========================================================
48
49 // No need to worry about closing streams: ifstream and ofstream close the stream automatically
50 // just before they get out of scope wherever they are
51
52 // Opens an input file and checks for anomalies
53 std::expected<std::ifstream, std::string> open_input_file(const std::string& filename) {
54 // Opens the input file in binary mode: the data in the file will be read as raw bytes
55 std::ifstream stream(filename, std::ios::binary);
56
57 if (!stream.is_open()) {
58 return std::unexpected(std::format("Error in opening input file '{}'", filename));
59 }
60
61 return stream;
62 }
63
64 // Opens an output file and checks for anomalies

66 // Opens the input file in binary mode: the data in the file will be read as raw bytes
67 std::ofstream stream(filename, std::ios::binary);
68 if (!stream.is_open()) {
69 return std::unexpected(std::format("Error in opening output file '{}'", filename));
70 }
71
