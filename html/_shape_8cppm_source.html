38export struct HitRecord {
39 Ray ray; // Ray that hit the shape
40 Point hit_point; // Point of intersection
41 Normal hit_normal; // Normal at the intersection point
42 Vec2D surface_params; // UV coordinates at the intersection point
43 float t; // Ray parameter at the intersection point
44
45 // std::shared_ptr<Shape> hitted_shape;
46 // GG: using raw pointer instead of the smart shared_ptr to avoid useless copies of the Shape
47 // HitRecord does not need to own the shape, it just needs a quick way to look at it
48 const Shape* hitted_shape = nullptr;
49
50 bool is_close(const HitRecord& other, float epsilon = 1e-5f) const; // Check if two HitRecords are close enough
51};
52
53bool HitRecord::is_close(const HitRecord& other, float epsilon) const {
54 return ray.is_close(other.ray, epsilon) &&
55 hit_point.is_close(other.hit_point, epsilon) &&
56 hit_normal.is_close(other.hit_normal, epsilon) &&
57 aux::are_close(surface_params.u, other.surface_params.u, epsilon) &&
58 aux::are_close(surface_params.v, other.surface_params.v, epsilon) &&
59 aux::are_close(t, other.t, epsilon) &&
60 hitted_shape == other.hitted_shape;
61}

81export struct Sphere : Shape {
82 using Shape::Shape; // Constructor takes in the Shape transformation
83
85 std::optional<HitRecord> ray_intersection(const Ray& ray) const { // Override method to compute ray-sphere intersection
86
87 // World Space -> Local Space
88 Ray local_ray = ray.transform(trans.inverse()); // Transform the ray to the sphere reference frame (where the sphere is a unit sphere centered at the origin)
89 // GG: to do so you need to use the inverse transformation!
90
91 // local_ray
92 Vec O = local_ray.origin.to_vec(); // ray origin is stored as a Vec, so it can be normalized
93 // (under the carpet the calculation is Point ray_origin - Point axes_origin = Vec ray_origin)
94 Vec D = local_ray.direction;
95 // tradeoff: if the most of the rays intersect the sphere, it's better to compute and
96 // store direction2 once instead of calling the method everytime.
97 float dir2 = D.norm2();
98
99 float b_half = O * D;
100 float discriminant = (b_half * b_half) - dir2 * (O.norm2() - 1.0f);
101
102 if (discriminant < 0.0f) {
103 return std::nullopt; // No intersection
104 }
105
106 float sqrt_disc = std::sqrt(discriminant);
107 float t1 = (-b_half - sqrt_disc) / dir2;
108 float t2 = (-b_half + sqrt_disc) / dir2;
109
110 // Find the closest valid t
111 float first_hit_t;
112
113 if (t1 > local_ray.tmin && t1 < local_ray.tmax) {
114 first_hit_t = t1;
115 } else if (t2 > local_ray.tmin && t2 < local_ray.tmax) {
116 first_hit_t = t2;
117 } else {
118 return std::nullopt; // Both t1 and t2 are out of bounds
119 }
120
121 // Local Space Geometry
122 Point local_point = local_ray.at(first_hit_t);
123
124 // Since the calculations assume to have a unit sphere centered at the origin
125 // the coordinates of the point of intersection are exactly the components of
126 // the normal to the sphere in that point
127 Normal local_normal{local_point.x, local_point.y, local_point.z};
128 if (local_point.to_vec() * D > 0.0f) {
129 local_normal = -local_normal;
130 }
131
132 // UV Coordinates MUST be calculated using the local normal.
133 // If you use the global normal, the texture won't rotate when you rotate the sphere!
134 float u = std::atan2(local_point.y, local_point.x) / (2.0f * std::numbers::pi_v<float>);
135 u = (u >= 0.0f) ? u : (u + 1.0f);
136 float v = std::acos(local_point.z) / std::numbers::pi_v<float>;
137
138 // Local Space -> World Space: report the HitRecord in the global space
139 HitRecord record;
140 record.ray = ray;
141 record.t = first_hit_t;
142 // Report point of intersection and normal in the global space
143 record.hit_point = trans * local_point;
144 // Normals transform with the transposed inverse: when operator * acts between a Transformation
145 // and a Normal it acts using the inverse transposed (see "Transformation of a Normal" in Geometry.cppm
146 // for reference).
147 // N.B.: if a scale transformation is applied the normal is no more normalized.
148 // Normalize it before return
149 record.hit_normal = (trans * local_normal).normalize();
150 record.surface_params = {u, v};
151
152 //record.hitted_shape = std::make_shared<Sphere>(*this); // Store a shared pointer to the hit shape in the record
153 // Point directly to this exact sphere in memory
154 record.hitted_shape = this;
155
156 return record;
157 }

164export struct Plane : Shape {
165 // Inherits constructors from Shape (allows passing Transformation)
166 using Shape::Shape;
167
168 [[nodiscard]] std::optional<HitRecord> ray_intersection(const Ray& ray) const override {
169
170 // Reference frame change: form World to local Shape frame
171 Ray local_ray = ray.transform(trans.inverse());
172
173 // Control parallelism: if z-component of ray direction is close to zero there is no intersection
174 if (std::abs(local_ray.direction.z) < 1e-5f) {
175 return std::nullopt;
176 }
177
178 // Intersection: O_z + t * d_z = 0 => t = -O_z / d_z
179 float t = -local_ray.origin.z / local_ray.direction.z;
180
181 // t limits control
182 if (t < local_ray.tmin || t > local_ray.tmax) {
183 return std::nullopt;
184 }
185
186 // Point of intersection in the frame of reference of the plane
187 Point local_point = local_ray.at(t);
188
189 // Return in the World space (apply direct transformation)
190 HitRecord record;
191 record.ray = ray;
192 record.t = t;
193 record.hit_point = trans * local_point;
194
195 Normal local_normal = (local_ray.direction.z < 0) ? Normal{0.0f, 0.0f, 1.0f} : Normal{0.0f, 0.0f, -1.0f};
196 record.hit_normal = trans * local_normal;
197
198 // Record (u, v) coordinates on the pair (Tile Pattern: represent infinite plane as composition of finite tiles)
199 // - floor(-3.2) = -4
200 record.surface_params.u = local_point.x - std::floor(local_point.x);
201 record.surface_params.v = local_point.y - std::floor(local_point.y);
202 //record.hitted_shape = std::make_shared<Plane>(*this); // Store a shared pointer to the hit shape in the record
203
204 // Point directly to this exact plane in memory
205 record.hitted_shape = this;
206
207 return record;
208 }

215export struct Cube : Shape {
216 using Shape::Shape;
217 [[nodiscard]] std::optional<HitRecord> ray_intersection(const Ray& ray) const override {
218 Ray local_ray = ray.transform(trans.inverse());
219
220 // Calculate the 1/dx, 1/dy and 1/dz only once
221 Vec inv_direction = Vec{
222 1.0f/local_ray.direction.x,
223 1.0f/local_ray.direction.y,
224 1.0f/local_ray.direction.z
225 };
226
227 // left-bottom-behind corner and right-upper-front corner of the canonical cube
228 Vec bounds[2] = {Vec{-1.0f, -1.0f, -1.0f}, Vec{1.0f, 1.0f, 1.0f}};
229
230 // This is used to determine which face gets hit first
231 int sign[3];
232 sign[0] = (local_ray.direction.x < 0) ? 1 : 0;
233 sign[1] = (local_ray.direction.y < 0) ? 1 : 0;
234 sign[2] = (local_ray.direction.z < 0) ? 1 : 0;
235
236 float tmin = (bounds[sign[0]].x - local_ray.origin.x) * inv_direction.x;
237 float tmax = (bounds[1-sign[0]].x - local_ray.origin.x) * inv_direction.x;
238 float tymin = (bounds[sign[1]].y - local_ray.origin.y) * inv_direction.y;
239 float tymax = (bounds[1-sign[1]].y - local_ray.origin.y) * inv_direction.y;
240
241 if (tmin > tymax || tymin > tmax) return std::nullopt;
242 if (tymin > tmin) tmin = tymin;
243 if (tymax < tmax) tmax = tymax;
244
245 float tzmin = (bounds[sign[2]].z - local_ray.origin.z) * inv_direction.z;
246 float tzmax = (bounds[1-sign[2]].z - local_ray.origin.z) * inv_direction.z;
247
248 if (tmin > tzmax || tzmin > tmax) return std::nullopt;
249 if (tzmin > tmin) tmin = tzmin;
250 if (tzmax < tmax) tmax = tzmax;
251
252 float first_hit_t{0.0f};
253 if (tmin > local_ray.tmin && tmin < local_ray.tmax) {
254 first_hit_t = tmin;
255 } else if (tmax > local_ray.tmin && tmax < local_ray.tmax) {
256 first_hit_t = tmax;
257 } else {
258 return std::nullopt;
259 }
260
261 Point local_point = local_ray.at(first_hit_t);
262 Normal local_normal{0.0f, 0.0f, 0.0f};
263
264 // To calculate the normal, evaluate which coordinate of the impact point is the biggest
265 // that will give the face on which the ray impacts and normal will be the normal vector to
266 // this face
267 float abs_x = std::abs(local_point.x);
268 float abs_y = std::abs(local_point.y);
269 float abs_z = std::abs(local_point.z);
270
271 if (abs_x >= abs_y && abs_x >= abs_z) {
272 local_normal = Normal{(local_point.x > 0.0f) ? 1.0f : -1.0f, 0.0f, 0.0f};
273 } else if (abs_y >= abs_x && abs_y >= abs_z) {
274 local_normal = Normal{0.0f, (local_point.y > 0.0f) ? 1.0f : -1.0f, 0.0f};
275 } else {
276 local_normal = Normal{0.0f, 0.0f, (local_point.z > 0.0f) ? 1.0f : -1.0f};
277 }
278
279 // Invert the normal if the ray comes from inside
280 if (local_point.to_vec() * local_ray.direction > 0.0f) {
281 local_normal = -local_normal;
282 }
283
284 // Coordinates on the single face of the cube
285 float u_local{0.0f}, v_local{0.0f};
286 // Coordinates in the Atlas representation
287 float u{0.0f}, v{0.0f};
288 // Auxiliary variables for the raw coordinates
289 float raw_u{0.0f}, raw_v{0.0f};
290 // Variables used to position the face in the atlas cross
291 float col{0.0f}, row{0.0f};
292
293 // u and v are taken as the free coordinates on the face of the cube that gets hit by the ray
294 // using Z-axis that points upward
295 // Afterwards u and v are transformed to map the faces of the cube in a cross
296
297 // one of the YZ faces
298 if (std::abs(local_normal.x) > 0.5f) {
299 // If the face is the one at positive X, Y-axis is oriented to the right, otherwise imagine
300 // to rotate the frame of reference and watch the face behind: the Y-axis now points to the left
301 raw_u = (local_point.x > 0.0f) ? local_point.y : -local_point.y;
302 raw_v = local_point.z;
303
304 // Set the position in the cross
305 col = 1.0f;
306 row = (local_point.x > 0.0f) ? 0.0f : 2.0f;
307 }
308 // one of the XZ faces
309 else if (std::abs(local_normal.y) > 0.5f) {
310 raw_u = (local_point.y > 0.0f) ? -local_point.x : local_point.x;
311 raw_v = local_point.z;
312
313 // Set the position in the cross
314 col = (local_point.y > 0.0f) ? 0.0f : 2.0f;
315 row = 2.0f;
316 }
317 // one of the XY faces
318 else if (std::abs(local_normal.z) > 0.5f) {
319 // Top face (+Z) or Bottom face (-Z) using Y-axis always right and X-axis downward for the
320 // top face and upward for the bottom face
321 // Climbing on the top face from the face X=-1 and going on the bottom face from X=-1
322 raw_u = -local_point.y;
323 raw_v = (local_point.z > 0.0f) ? local_point.x : -local_point.x;
324
325 // Set the position in the cross
326 col = 1.0f;
327 row = (local_point.z > 0.0f) ? 3.0f : 1.0f;
328 }
329
330 // Convert the local coordinates from [-1, 1] range to standard UV [0, 1] range
331 u_local = (raw_u + 1.0f) * 0.5f;
332 v_local = (raw_v + 1.0f) * 0.5f;
333
334 u = (col + u_local)/3.0f;
335 v = (row + v_local)/4.0f;
336
337
338 HitRecord record;
339 record.ray = ray;
340 record.t = first_hit_t;
341 record.hit_point = trans * local_point;
342 record.hit_normal = (trans * local_normal).normalize();
343 record.surface_params.u = u;
344 record.surface_params.v = v;
345 record.hitted_shape = this;
346
347 return record;
348 }
349};

356export struct World {
357 std::vector<std::unique_ptr<Shape>> shapes;
358 // Add a shape to the World: use unique_ptr + move pattern for optimization
359 void add(std::unique_ptr<Shape> shape) {
360 shapes.push_back(std::move(shape));
361 }
362
363 [[nodiscard]] std::optional<HitRecord> ray_intersection(Ray ray) const {
364 // Default: no hit
365 std::optional<HitRecord> closest = std::nullopt;
366
367 // GG: I find passing by copy is the best way to go. Passing it by reference, the original ray gets
368 // permanently modified making it harder to debug if tmax has changed
369
370 // Cycle on shapes in the scene
371 for (const auto& shape : shapes) {
372 if (auto intersection = shape->ray_intersection(ray)){
373 closest = intersection;
374 ray.tmax = closest->t; // Update tmax to the closest hit
375 // RP: I'll just change the tmax! In this way I avoid to actually generate
376 // the HitRecord object and then throw it away.
377 // If we hit we also update both closest and tmax.
378 // If we never hit a shape, there's the default std::nullopt return.
379 // GG: This is great!
380 }
381 }
382 return closest;
383 }
