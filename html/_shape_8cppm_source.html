74bool HitRecord::is_close(const HitRecord& other, float epsilon) const {
75 return ray.is_close(other.ray, epsilon) &&
76 hit_point.is_close(other.hit_point, epsilon) &&
77 hit_normal.is_close(other.hit_normal, epsilon) &&
78 aux::are_close(surface_params.u, other.surface_params.u, epsilon) &&
79 aux::are_close(surface_params.v, other.surface_params.v, epsilon) &&
80 aux::are_close(t, other.t, epsilon) &&
81 hitted_shape == other.hitted_shape;
82}

125export struct Sphere : Shape {
126 using Shape::Shape; // Constructor takes in the Shape transformation
127
129 std::optional<HitRecord> ray_intersection(const Ray& ray) const { // Override method to compute ray-sphere intersection
130
131 // Transform the ray to the sphere reference frame (where the sphere is a unit sphere centered at the origin)
132 Ray local_ray = ray.transform(trans.inverse());
133
134 Vec O = local_ray.origin.to_vec(); // ray origin is stored as a Vec, so it can be normalized
135 // (under the carpet the calculation is Point ray_origin - Point axes_origin = Vec ray_origin)
136 Vec D = local_ray.direction;
137
138 float dir2 = D.norm2();
139
140 float b_half = O * D;
141 float discriminant = (b_half * b_half) - dir2 * (O.norm2() - 1.0f);
142
143 if (discriminant < 0.0f) {
144 return std::nullopt;
145 }
146
147 float sqrt_disc = std::sqrt(discriminant);
148 float t1 = (-b_half - sqrt_disc) / dir2;
149 float t2 = (-b_half + sqrt_disc) / dir2;
150
151 // Find the closest valid t
152 float first_hit_t;
153
154 if (t1 > local_ray.tmin && t1 < local_ray.tmax) {
155 first_hit_t = t1;
156 } else if (t2 > local_ray.tmin && t2 < local_ray.tmax) {
157 first_hit_t = t2;
158 } else {
159 return std::nullopt; // Both t1 and t2 are out of bounds
160 }
161
162 // Local Space Geometry
163 Point local_point = local_ray.at(first_hit_t);
164
165 // Since the calculations assume to have a unit sphere centered at the origin the coordinates of the point of
166 // intersection are exactly the components of the normal to the sphere in that point
167 Normal local_normal{local_point.x, local_point.y, local_point.z};
168 if (local_point.to_vec() * D > 0.0f) {
169 local_normal = -local_normal;
170 }
171
172 // Compute texture coordinates in local space so that textures follow object transformations.
173 float u = std::atan2(local_point.y, local_point.x) / (2.0f * std::numbers::pi_v<float>);
174 u = (u >= 0.0f) ? u : (u + 1.0f);
175 float v = std::acos(local_point.z) / std::numbers::pi_v<float>;
176
177 // Local Space -> World Space: report the HitRecord in the global space
178 HitRecord record;
179 record.ray = ray;
180 record.t = first_hit_t;
181 record.hit_point = trans * local_point;
182
183 // Normals are transformed using the inverse transpose transformation.
184 // Renormalize after transformation to account for scaling.
185 record.hit_normal = (trans * local_normal).normalize();
186 record.surface_params = {u, v};
187 record.hitted_shape = this;
188
189 return record;
190 }
191
192};

201export struct Plane : Shape {
202
203 using Shape::Shape; // Inherits constructors from Shape (allows passing Transformation)
204
205 [[nodiscard]] std::optional<HitRecord> ray_intersection(const Ray& ray) const override {
206
207 // Reference frame change: form World to local Shape frame
208 Ray local_ray = ray.transform(trans.inverse());
209
210 // Control parallelism: if z-component of ray direction is close to zero there is no intersection
211 if (std::abs(local_ray.direction.z) < 1e-5f) {
212 return std::nullopt;
213 }
214
215 // Intersection: O_z + t * d_z = 0 => t = -O_z / d_z
216 float t = -local_ray.origin.z / local_ray.direction.z;
217
218 // t limits control
219 if (t < local_ray.tmin || t > local_ray.tmax) {
220 return std::nullopt;
221 }
222
223 // Return geometric information for intersection in global space
224 Point local_point = local_ray.at(t);
225 HitRecord record;
226 record.ray = ray;
227 record.t = t;
228 record.hit_point = trans * local_point;
229 Normal local_normal = (local_ray.direction.z < 0) ? Normal{0.0f, 0.0f, 1.0f} : Normal{0.0f, 0.0f, -1.0f};
230 record.hit_normal = trans * local_normal;
231
232 // Generate repeating texture coordinates from local position.
233 record.surface_params.u = local_point.x - std::floor(local_point.x);
234 record.surface_params.v = local_point.y - std::floor(local_point.y);
235 record.hitted_shape = this;
236
237 return record;
238 }
239};

251export struct Cube : Shape {
252 using Shape::Shape;
253 [[nodiscard]] std::optional<HitRecord> ray_intersection(const Ray& ray) const override {
254
255 // Transform the ray into the cube's local reference frame.
256 Ray local_ray = ray.transform(trans.inverse());
257
258 // Ray-box intersection using the slab method.
259 Vec inv_direction = Vec{
260 1.0f/local_ray.direction.x,
261 1.0f/local_ray.direction.y,
262 1.0f/local_ray.direction.z
263 };
264
265 // left-bottom-behind corner and right-upper-front corner of the canonical cube
266 Vec bounds[2] = {Vec{-1.0f, -1.0f, -1.0f}, Vec{1.0f, 1.0f, 1.0f}};
267
268 // Determine which side of each slab is encountered first.
269 int sign[3];
270 sign[0] = (local_ray.direction.x < 0) ? 1 : 0;
271 sign[1] = (local_ray.direction.y < 0) ? 1 : 0;
272 sign[2] = (local_ray.direction.z < 0) ? 1 : 0;
273
274 float tmin = (bounds[sign[0]].x - local_ray.origin.x) * inv_direction.x;
275 float tmax = (bounds[1-sign[0]].x - local_ray.origin.x) * inv_direction.x;
276 float tymin = (bounds[sign[1]].y - local_ray.origin.y) * inv_direction.y;
277 float tymax = (bounds[1-sign[1]].y - local_ray.origin.y) * inv_direction.y;
278
279 if (tmin > tymax || tymin > tmax) return std::nullopt;
280 if (tymin > tmin) tmin = tymin;
281 if (tymax < tmax) tmax = tymax;
282
283 float tzmin = (bounds[sign[2]].z - local_ray.origin.z) * inv_direction.z;
284 float tzmax = (bounds[1-sign[2]].z - local_ray.origin.z) * inv_direction.z;
285
286 if (tmin > tzmax || tzmin > tmax) return std::nullopt;
287 if (tzmin > tmin) tmin = tzmin;
288 if (tzmax < tmax) tmax = tzmax;
289
290 // Choose the first valid intersection along the ray.
291 float first_hit_t{0.0f};
292 if (tmin > local_ray.tmin && tmin < local_ray.tmax) {
293 first_hit_t = tmin;
294 } else if (tmax > local_ray.tmin && tmax < local_ray.tmax) {
295 first_hit_t = tmax;
296 } else {
297 return std::nullopt;
298 }
299
300 Point local_point = local_ray.at(first_hit_t);
301 Normal local_normal{0.0f, 0.0f, 0.0f};
302
303 // The dominant coordinate identifies the intersected face.
304 float abs_x = std::abs(local_point.x);
305 float abs_y = std::abs(local_point.y);
306 float abs_z = std::abs(local_point.z);
307
308 if (abs_x >= abs_y && abs_x >= abs_z) {
309 local_normal = Normal{(local_point.x > 0.0f) ? 1.0f : -1.0f, 0.0f, 0.0f};
310 } else if (abs_y >= abs_x && abs_y >= abs_z) {
311 local_normal = Normal{0.0f, (local_point.y > 0.0f) ? 1.0f : -1.0f, 0.0f};
312 } else {
313 local_normal = Normal{0.0f, 0.0f, (local_point.z > 0.0f) ? 1.0f : -1.0f};
314 }
315
316 // Flip the normal for rays originating inside the cube.
317 if (local_point.to_vec() * local_ray.direction > 0.0f) {
318 local_normal = -local_normal;
319 }
320
321 /* ---------------------------------------------------------
322 * Cube atlas UV mapping
323 * The hit point is first converted into local coordinates
324 * on the intersected face. The six faces are then arranged
325 * into a cross-shaped texture atlas.
326 ---------------------------------------------------------- */
327 // Coordinates on the single face of the cube
328 float u_local{0.0f}, v_local{0.0f};
329 // Coordinates in the Atlas representation
330 float u{0.0f}, v{0.0f};
331 // Auxiliary variables for the raw coordinates
332 float raw_u{0.0f}, raw_v{0.0f};
333 // Variables used to position the face in the atlas cross
334 float col{0.0f}, row{0.0f};
335
336
337 // YZ face (+X or -X)
338 if (std::abs(local_normal.x) > 0.5f) {
339 // Compute local face coordinates with a consistent orientation.
340 raw_u = (local_point.x > 0.0f) ? local_point.y : -local_point.y;
341 raw_v = local_point.z;
342
343 // Set the position in the cross
344 col = 1.0f;
345 row = (local_point.x > 0.0f) ? 0.0f : 2.0f;
346 }
347 // XZ face (+Y or -Y)
348 else if (std::abs(local_normal.y) > 0.5f) {
349 raw_u = (local_point.y > 0.0f) ? -local_point.x : local_point.x;
350 raw_v = local_point.z;
351
352 // Set the position in the cross
353 col = (local_point.y > 0.0f) ? 0.0f : 2.0f;
354 row = 2.0f;
355 }
356 // XY face (+Z or -Z)
357 else if (std::abs(local_normal.z) > 0.5f) {
358 // Top (+Z) and bottom (-Z) faces.
359 //
360 // UV coordinates are chosen so that neighbouring faces share a consistent orientation when
361 // unfolded into the cube-map cross atlas.
362 raw_u = -local_point.y;
363 raw_v = (local_point.z > 0.0f) ? local_point.x : -local_point.x;
364
365 // Set the position in the cross
366 col = 1.0f;
367 row = (local_point.z > 0.0f) ? 3.0f : 1.0f;
368 }
369
370 // Convert the local coordinates from [-1, 1] range to standard UV [0, 1] range
371 u_local = (raw_u + 1.0f) * 0.5f;
372 v_local = (raw_v + 1.0f) * 0.5f;
373
374 u = (col + u_local)/3.0f;
375 v = (row + v_local)/4.0f;
376
377
378 // Geometric information on the intersection point in global space
379 HitRecord record;
380 record.ray = ray;
381 record.t = first_hit_t;
382 record.hit_point = trans * local_point;
383 record.hit_normal = (trans * local_normal).normalize();
384 record.surface_params.u = u;
385 record.surface_params.v = v;
386 record.hitted_shape = this;
387
388 return record;
389 }
390};

402export struct World {
403
405 std::vector<std::unique_ptr<Shape>> shapes;
406
412 void add(std::unique_ptr<Shape> shape) {
413 shapes.push_back(std::move(shape));
414 }
415
422 [[nodiscard]] std::optional<HitRecord> ray_intersection(Ray ray) const {
423 // Default: no hit
424 std::optional<HitRecord> closest = std::nullopt;
425
426 // Cycle on shapes in the scene
427 for (const auto& shape : shapes) {
428 if (auto intersection = shape->ray_intersection(ray)){
429 closest = intersection;
430
431 // Update the maximum ray distance so that only closer intersections can be accepted.
432 ray.tmax = closest->t;
433 }
434 }
435 return closest;
436 }
