27 PCG(std::uint64_t init_state = 42, std::uint64_t init_seq = 54) {
28 state = 0;
29 inc = (init_seq << 1) | 1u;
30 random();

35 // It returns a random 32 bit integer
36 std::uint32_t random() {
37 std::uint64_t oldstate = state;
38 state = oldstate * 6364136223846793005ULL + inc;
39
40 // Apply the scrambling function (XorShift + RandomRotation)
41 std::uint32_t xorshifted = static_cast<std::uint32_t>(((oldstate >> 18u) ^ oldstate) >> 27u);
42 std::uint32_t rot = static_cast<std::uint32_t>(oldstate >> 59u);
43
44 // Using the C++20/23 rotr rotation function
45 return std::rotr(xorshifted, rot);
46 }
47
48 // It returns a uniform RGN in [0, 1)
49 float random_float() {
50 return static_cast<float>(random()) / 4294967296.0f;
51 }
