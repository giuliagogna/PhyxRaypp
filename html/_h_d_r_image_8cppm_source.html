115 // Support function to check endianness: if the endianness of the file and the native OS endianness
116 // are different, it swaps the bytes order
117 static uint32_t _swap_if_needed(const uint32_t raw_bytes, Endianness file_endianness) {
118 bool is_file_little = (file_endianness == Endianness::little_endian);
119 bool is_native_little = (std::endian::native == std::endian::little);
120
121 if (is_file_little != is_native_little) {
122 return std::byteswap(raw_bytes); // efficiently swaps the byte order of the 32-bit integer
123 }
124
125 return raw_bytes;
126 }

163 // Reads a sequence of 4 bytes from a binary stream and converts it to a float.
164 // This function is designed to be robust against various file format issues, such as truncated files or invalid data.
165 static std::expected<float, InvalidPfmFileFormat> _read_float(std::istream& stream, Endianness file_endianness) {
166 uint32_t raw_bytes{};
167
168 // Reading 4 bytes from the stream into raw_bytes.
169 stream.read(reinterpret_cast<char*>(&raw_bytes), sizeof(raw_bytes));
170
171 // catches file faults and end of file.
172 if (stream.fail()) {
173 if (stream.gcount() > 0 && stream.gcount() < 4) {
174 return std::unexpected(InvalidPfmFileFormat{"Truncated file: expected 4 bytes for a float but only read " +
175 std::to_string(stream.gcount()) + " bytes."});
176 }
177 return std::unexpected(InvalidPfmFileFormat{"Reading error: unable to read 4 bytes for a float."});
178 }
179
180 // Endianness handling
181 raw_bytes = _swap_if_needed(raw_bytes, file_endianness);
182
183 // Interpret raw bites with corrrect endianness as a float and return it
184 return std::bit_cast<float>(raw_bytes);
185 }

187 // Reads a line in the header, skipping comments (lines starting with #) and empty lines.
188 static std::expected<std::string, InvalidPfmFileFormat> _read_line(std::istream& stream) {
189 std::string result;
190
191 // read till we find a line that is not a comment (starting with #) and not empty (for example an extra newline)
192 while (std::getline(stream, result)) {
193 // remove carriage return \r
194 if (!result.empty() && result.ends_with('\r')) {
195 result.pop_back();
196 }
197
198 // if the line starts with # it is a comment and we skip it (flexibility)
199 if (!result.empty() && result.starts_with('#')) {
200 continue;
201 }
202
203 // if the line is empty we skip it (for example an extra newline) (flexibility)
204 if (result.empty()) {
205 continue;
206 }
207
208 return result;
209 }
210
211 return std::unexpected(InvalidPfmFileFormat{"Impossible to read line."});
212 }

214 // Converts the endianness line to the correspondent Endianness type checking the sign.
215 // At this point the file is not binary yet, we are still reading the header that is in ASCII, so there is no need to read float
216 // values with _read_float.
217
218 static std::expected<Endianness, InvalidPfmFileFormat> _parse_endianness(const std::string& line) {
219 std::istringstream stream(line);
220 float value;
221
222 // If the line that should contain the endianness contains something that is not a number, raises an error
223 if (!(stream >> value)) {
224 return std::unexpected(InvalidPfmFileFormat{"Invalid endianness format. Read: '" + line + "'."});
225 }
226
227 // Evaluate only the sign of the value read in the line: if positive little_endian, if negative big_endian
228 if (value > 0.0f) {
229 return Endianness::big_endian;
230 } else if (value < 0.0f) {
231 return Endianness::little_endian;
232 } else {
233 // If the value is zero it cannot be interpreted as an endianness
234 return std::unexpected(InvalidPfmFileFormat{"Invalid endianness. Value cannot be zero. Read: '" + line + "'."});
235 }
236 }

238 // Parses the image dimensions from the size line, ensuring correct format and valid values.
239 static std::expected<std::pair<int, int>, InvalidPfmFileFormat> _parse_img_size(const std::string& line) {
240 std::istringstream stream(line);
241 int parsed_width{}, parsed_height{};
242
243 // stream >> extracts the first value, skips the spacing and extracts the second value
244 // the fact that stream >> skips the spaces is useful because our values may be separated ny more than one
245 // 1 space or be followed by white spaces
246
247 // If one tries to read a first number that is not an integer like 10.1, the first >> will read 10 and
248 // put it into parsed_width.
249 // The second >> will encounter . that is not a number and it will raise the unexpected value
250 if (!(stream >> parsed_width >> parsed_height)) {
251 return std::unexpected(InvalidPfmFileFormat{"Cannot extract image dimensions from line. Read: " + line});
252 }
253
254 // If one tries to read second number that is not an integer like 10 20.1
255 // The first >> will read 10 and put it into parsed_width
256 // The second >> will read 20 and put it into parsed_height
257 // Then this third stream will be called and it will encounter . and therefore it will raise an error for
258 // too many values
259
260 stream >> std::ws; // eliminates the remaining white spaces
261 // If after all the remaining white spaces we have not found the end of the line it means that there are too
262 // many values
263 if (!stream.eof()) {
264 return std::unexpected(InvalidPfmFileFormat{"Too many values for dimensions. Invalid format: " + line});
265 }
266
267 // Here the sign of the numbers read are controlled
268 if (parsed_width <= 0 || parsed_height <= 0) {
269 return std::unexpected(InvalidPfmFileFormat{"Image dimensions must be greater than zero. Read: " + line});
270 }
271
272 return std::make_pair(parsed_width, parsed_height);
273 }

279 // Reads a PFM from stream. Handles the entire reading process, including format validation, dimension parsing, endianness handling, and pixel data extraction.
280 // Errors are propagated using std::expected, allowing the caller to handle them gracefully.
281 static std::expected<HDRImage, InvalidPfmFileFormat> read_pfm_file(std::istream& stream) {
282 auto format = _read_line(stream);
283 if (!format.has_value()) { // if the line that should contain the format cannot be read, return the error
284 return std::unexpected(format.error());
285 }
286 if (format.value() != "PF") {
287 return std::unexpected(InvalidPfmFileFormat{"Non valid format. Expected 'PF' read " + format.value()});
288 }
289
290 // Image size line
291 auto size_line = _read_line(stream);
292 if (!size_line.has_value()) return std::unexpected(size_line.error());
293 auto size_res = _parse_img_size(size_line.value());
294 if (!size_res.has_value()) return std::unexpected(size_res.error());
295 int width = size_res.value().first;
296 int height = size_res.value().second;
297
298 // Endianness line
299 auto endian_line_res = _read_line(stream);
300 if (!endian_line_res.has_value()) return std::unexpected(endian_line_res.error());
301 auto endian_res = _parse_endianness(endian_line_res.value());
302 if (!endian_res.has_value()) return std::unexpected(endian_res.error());
303
304 Endianness endianness = endian_res.value();
305
306 HDRImage img(width, height);
307
308 for (int y = height - 1; y >= 0; --y) {
309 for (int x = 0; x < width; ++x) {
310 auto r_res = _read_float(stream, endianness);
311 if (!r_res.has_value()) return std::unexpected(r_res.error());
312
313 auto g_res = _read_float(stream, endianness);
314 if (!g_res.has_value()) return std::unexpected(g_res.error());
315
316 auto b_res = _read_float(stream, endianness);
317 if (!b_res.has_value()) return std::unexpected(b_res.error());
318
319 img.set_pixel(x, y, Color(r_res.value(), g_res.value(), b_res.value()));
320 }
321 }
322
323 // Consume final white spaces or end of line characters (\n, \r, ' ')
324 stream >> std::ws;
325
326 // Try to read the next character
327 if (!stream.eof()) {
328 // If the file is not at the end return this error.
329 // GG: I don't like this being an error, would like to make it a warning
330 return std::unexpected(InvalidPfmFileFormat{"Unexpected data after reading all pixels."});
331 }
332
333 // Returned the finished image
334 return img;
335 }

356 // Safely converts a float to a 4-byte sequence and writes it to a binary stream.
357 static void _write_float(std::ostream& stream, float value, Endianness file_endianness) {
358 // Copy the float's exact bit pattern into an unsigned 32-bit integer.
359 // Here the sequence of the 4 byte is in the order imposed by the native endianness of the OS
360 uint32_t raw_bytes = std::bit_cast<uint32_t>(value);
361
362 // If the native endianness differs from the endianness I want my file to be written in
363 // I need to swap the bytes
364 raw_bytes = _swap_if_needed(raw_bytes, file_endianness);
365
366 // When the bytes have the order that I want based on the endianness I have control on
367 // I can write them on the stream
368 stream.write(reinterpret_cast<const char*>(&raw_bytes), sizeof(raw_bytes));
369
370 // This way when I declare the endianness in the file I am sure that my float values are represented
371 // with the correct endianness
372 }
373
374 // Writes a pfm on a stream
375 // [[nodiscard]]: if a user tries to use it ignoring its return value the compiler gives a warning: if
376 // something bad happens, if the error value is not caught in the main the program proceeds. with [[nodiscard]]
377 // the user is warned to catch the return value of this function in the main
378 [[nodiscard]] std::expected<void, std::string> write_pfm_file(std::ostream& stream, Endianness file_endianness) const {
379
380 // Endianness line
381 std::string endianness_string = (file_endianness == Endianness::little_endian) ? "-1.0" : "1.0";
382
383 stream << "PF\n" << width << " " << height << "\n" << endianness_string << "\n";
384
385 // If there was a problem in writing the header catch it: when stream << encounters a problem it raises
386 // a red light and all the following writing operations are ignored. At the end stream.good() checks
387 // if the red light was turned on and reports it.
388 if (!stream.good()) {
389 return std::unexpected(std::format("Error in writing header. Memory full or disconnected."));
390 }
391
392 for (int y = height -1; y >= 0; --y) {
393 for (int x = 0; x < width; ++x) {
394 Color c = get_pixel(x, y);
395 _write_float(stream, c.r, file_endianness);
396 _write_float(stream, c.g, file_endianness);
397 _write_float(stream, c.b, file_endianness);
398 }
399 }
400
401 // If there was a problem in writing the pixel colors catch it: just check it at the end of the for loop
402 // If I put a check inside the _write_float function the processor needs to check an if condition at every
403 // iteration compromising performance.
404 // Checking at the end of the stream is safe enough because if something went wrong at some point
405 // the red light is on and .good() reports it
406 if (!stream.good()) {
407 return std::unexpected(std::format("Error in writing pixel colors. Memory full or disconnected."));
408 }
409
410 return {};
411 }

413 // Provide file name and endianness, and this function will handle the entire writing process, including file opening,
414 // header writing, pixel data streaming, and error handling with detailed messages.
415 [[nodiscard]] std::expected<void, std::string> write_pfm_file(const std::string& filename, Endianness file_endianness) const {
416
417 auto stream_res = aux::open_output_file(filename);
418
419 if (!stream_res.has_value()) {
420 return std::unexpected(stream_res.error());

429 }

431 // Return void to signal complete success

471 // Computes the average luminosity of the picture according to Shirley & Morley (2003) algorithm
472 // Passing from log10 to log2 which is more direct. Since these two logarithms differ by a constant factor log2(10), the final result is the same.
473 // pf: log2(x) = log10(x) / log10(2) => sum log2(x) = sum log10(x) / log10(2) => 10^(sum log10(x) / N) = 2^(sum log2(x) / N)
474 [[nodiscard]] std::expected<float, std::string> average_luminosity(float delta = 1e-10, std::string luminosity_type = "bt709") const {
475
476 float cumsum = 0.0f;
477 float tolerance = 1e-6f; // tolerance for negative luminosity values that may arise from rounding errors.
478
479 int length = pixels.size();
480 if (!length) {
481 return std::unexpected("Cannot compute average luminosity of an empty image.");
482 }
483
484 if (delta <= 0.0f) {
485 return std::unexpected("Delta value must be strictly greater than zero to avoid logarithm of zero or negative numbers. Received: " + std::to_string(delta));
486 }
487
488 float current_pixel_luminosity{};
489
490 // Code duplication, I know, but I don't want to check which is the method via string each time...
491 if (luminosity_type == "mid_range") {
492 for (auto& pixel : pixels) {
493 current_pixel_luminosity = pixel.luminosity_mid_range();
494 if (current_pixel_luminosity < -tolerance) {
495 return std::unexpected("Negative luminosity value encountered in mid_range method. Luminosity values must be non-negative. Check the pixel values in the image.");
496 }
497 // If pixel is negative but greater than tolerance, set it to 0.0f: does not enter here if it entered the error
498 if (current_pixel_luminosity < 0.0f) current_pixel_luminosity = 0.0f;
499 cumsum += log2(current_pixel_luminosity + delta);
500 }
501 } else if (luminosity_type == "arithmetic_mean") {
502 for (auto& pixel : pixels) {
503 current_pixel_luminosity = pixel.luminosity_arithmetic_mean();
504 if (current_pixel_luminosity < -tolerance) {
505 return std::unexpected("Negative luminosity value encountered in arithmetic_mean method. Luminosity values must be non-negative. Check the pixel values in the image.");
506 }
507 // If pixel is negative but greater than tolerance, set it to 0.0f: does not enter here if it entered the error
508 if (current_pixel_luminosity < 0.0f) current_pixel_luminosity = 0.0f;
509 cumsum += log2(current_pixel_luminosity + delta);
510 }
511 } else if (luminosity_type == "bt709") {
512 for (auto& pixel : pixels) {
513 current_pixel_luminosity = pixel.luminosity_bt709();
514 if (current_pixel_luminosity < -tolerance) {
515 return std::unexpected("Negative luminosity value encountered in bt709 method. Luminosity values must be non-negative. Check the pixel values in the image.");
516 }
517 // If pixel is negative but greater than tolerance, set it to 0.0f: does not enter here if it entered the error
518 if (current_pixel_luminosity < 0.0f) current_pixel_luminosity = 0.0f;
519 cumsum += log2(current_pixel_luminosity + delta);
520 }
521 } else {
522 return std::unexpected("Invalid luminosity type. Expected 'mid_range', 'arithmetic_mean' or 'bt709'. Received: " + luminosity_type);
523 }
524
525 return pow(2.0f, cumsum / length);
526 }

554 // Renormalization of the image luminosity using a manual value
555 std::expected<void, std::string> normalize_image(float normalization, float luminosity) {
