334 void skip_whitespaces_and_comments() {
335
336 std::optional<char> ch = read_char();
337
338 while (ch.has_value() && (ch.value() == ' ' ||
339 ch.value() == '\t' ||
340 ch.value() == '\n' ||
341 ch.value() == '\r' ||
342 ch.value() == '#')) {
343
344 if (ch.value() == '#') {
345 std::optional<char> comment_ch = read_char();
346 while (comment_ch.has_value() && comment_ch != '\r' && comment_ch != '\n') {
347 comment_ch = read_char();
348 }
349 }
350
351 ch = read_char();
352 if (!ch.has_value()) {
353 return;
354 }
355 }
356
357 unread_char(ch);

376 std::expected<std::unique_ptr<Token>, GrammarError>_parse_string_token(SourceLocation token_location) {
377 std::string res_string = "";
378 std::optional<char> ch = read_char();
379
380 while (ch.has_value() && ch.value() != '"') {
381 res_string.push_back(ch.value());
382 ch = read_char();
383 }
384
385 if (!ch.has_value()) {
386 // It ended because we ran out of characters. Error!
387 return std::unexpected(GrammarError{token_location, "Unterminated string"});
388 }
389
390 return std::make_unique<LiteralStringToken>(token_location, res_string);
391 }

410 std::expected<std::unique_ptr<Token>, GrammarError> _parse_float_token(char first_char, SourceLocation token_location) {
411 std::string float_string = "";
412 float_string.push_back(first_char);
413
414 std::optional<char> ch = read_char();
415
416 while (ch.has_value() && (std::isdigit(ch.value()) ||
417 ch.value() == '.' ||
418 ch.value() == 'e' ||
419 ch.value() == 'E' ||
420 ch.value() == '+' ||
421 ch.value() == '-')) {
422
423 float_string.push_back(ch.value());
424 ch = read_char();
425 }
426
427 // If the loop finished and 'ch' still has a value, it means we hit a
428 // character that isn't part of a number (like a comma or space). Put it back.
429 if (ch.has_value()) {
430 unread_char(ch);
431 }
432
433 float res_float;
434 char* end_ptr = nullptr;
435
436 errno = 0;
437
438 res_float = std::strtof(float_string.c_str(), &end_ptr);
439
440 // Check if the number was too big or too small to fit in a float
441 if (errno == ERANGE) {
442 return std::unexpected(GrammarError{
443 token_location,
444 std::format("'{}' is out of the floating-point value range.", float_string)
445 });
446 }
447
448 // If the end_ptr didn't move from the start, it completely failed to parse a number
449 if (end_ptr == float_string.c_str()) {
450 return std::unexpected(GrammarError{
451 token_location,
452 std::format("'{}' is an invalid floating-point number", float_string)
453 });
454 }
455
456 return std::make_unique<LiteralNumberToken>(token_location, res_float);
457
458 }

476 std::unique_ptr<Token> _parse_keyword_or_identifier_token(char first_char, SourceLocation token_location) {
477 std::string res_string = "";
478 res_string.push_back(first_char);
479
480 std::optional<char> ch = read_char();
481
482 while (ch.has_value() && (std::isalnum(ch.value()) || ch.value() == '_')) {
483 res_string.push_back(ch.value());
484 ch = read_char();
485 }
486
487 // If the loop finished and 'ch' still has a value, it means we hit a
488 // character that isn't alphanumeric or underscore. Put it back.
489 if (ch.has_value()) {
490 unread_char(ch);
491 }
492
493 if (KEYWORDS.contains(res_string)) {
494 return std::make_unique<KeywordToken>(token_location, KEYWORDS.at(res_string));
495 } else {
496 return std::make_unique<IdentifierToken>(token_location, res_string);
497 }
498
499 }

524 std::expected<std::unique_ptr<Token>, GrammarError> read_token() {
525
526 if (saved_token.has_value()) {
527 // Return the token stored by unread_token() without reading from the stream.
528 auto result = std::move(saved_token.value());
529 saved_token = std::nullopt;
530 return result;
531 }
532
533 skip_whitespaces_and_comments();
534
535 std::optional<char> ch = read_char();
536 if (!ch.has_value()) {
537 return std::make_unique<StopToken>(location);
538 }
539
540 SourceLocation token_location = location;
541
542 if (SYMBOLS.contains(ch.value())) {
543 // One character symbol like '(', ']', '>'
544 return std::make_unique<SymbolToken>(token_location, ch.value());
545 } else if (ch.value() == '"') {
546 // Here starts a literal string
547 return _parse_string_token(token_location);
548 } else if (std::isdigit(ch.value()) || ch.value() == '.' || ch.value() == '+' || ch.value() == '-') {
549 // Check if the character is the start of a float number
550 return _parse_float_token(ch.value(), token_location);
551 } else if (std::isalpha(ch.value()) || ch.value() == '_') {
552 // Since it begins with an alphabetic character, it must either be a keyword or a identifier
553 return _parse_keyword_or_identifier_token(ch.value(), token_location);
554 } else {
555 // We have an invalid character as '@' or '&'
556 return std::unexpected(GrammarError{location, std::format("Invalid character '{}'", ch.value())});
557 }
558 }

575 assert(!saved_token.has_value());
576 saved_token = std::move(token);
577 }

634 std::expected<void, GrammarError> expect_symbol(InputStream& input_file, char symbol) {
635
636 auto res = input_file.read_token();
637 if (!res.has_value()) {
638 return std::unexpected(res.error());
639 }
640 std::unique_ptr<Token> token = std::move(res.value());
641
642 auto* sym_token = dynamic_cast<SymbolToken*>(token.get());
643
644 // If what comes next is not a Symbol token, throw the GrammarError
645 if (sym_token == nullptr) {
646 return std::unexpected(GrammarError{
647 token->location,
648 std::format("Expected symbol '{}', but got a different token type", symbol)
649 });
650 }
651 // If it's the wrong symbol, throw the GrammarError
652 if (sym_token->symbol != symbol) {
653 return std::unexpected(GrammarError{
654 token->location,
655 std::format("Expected symbol '{}', but got '{}'", symbol, sym_token->symbol)
656 });
657 }
658
659 return {};
660 }
