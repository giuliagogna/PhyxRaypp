259// Generate an orthogonal ray in camera space and transform it to world space.
260Ray OrthogonalCamera::fire_ray(const float u, const float v) const {
261 const Point ray_origin{-1.0f, (1.0f - 2.0f * u) * aspect_ratio, 2.0f * v - 1.0f};
262 constexpr Vec ray_direction{1.0f, 0.0f, 0.0f};
263 return Ray{ray_origin, ray_direction}.transform(trans);
264}

266// Generate a perspective ray in camera space and transform it to world space
267Ray PerspectiveCamera::fire_ray(const float u, const float v) const {
268 const Point ray_origin{-d, 0.0f, 0.0f};
269 const Vec ray_direction{d, (1.0f - 2.0f * u) * aspect_ratio, 2.0f * v - 1.0f};
270 return Ray{ray_origin, ray_direction}.transform(trans);

273// Convert pixel coordinates to normalized image-plane coordinates.
274Ray ImageTracer::fire_ray(const int row, const int col, const float u_pixel, const float v_pixel) const {
275 const float width = static_cast<const float>(frame.width);
276 const float height = static_cast<const float>(frame.height);
277
278 const float u = (static_cast<const float>(col) + u_pixel) / width;
279 const float v = 1.0f - (static_cast<const float>(row) + v_pixel) / height;
280
281 return camera.fire_ray(u, v);
282}

287void ImageTracer::fire_all_rays(const std::function<Color(const Ray&)>& func) {
288 const int width = frame.width;
289 const int height = frame.height;
290
291 // Render using a single sample per pixel.
292 for (int row = 0; row < height; ++row) {
293 for (int col = 0; col < width; ++col) {
294 Ray ray = fire_ray(row, col, 0.5f, 0.5f);
295 Color color = func(ray);
296 frame.set_pixel(col, row, color);
297 }
298 }
299}

301void ImageTracer::fire_all_rays(const std::function<Color(const Ray&)>& func, PCG& pcg, int antialiasing_level) {
302 const int width = frame.width;
303 const int height = frame.height;
304 if (antialiasing_level < 1) antialiasing_level = 1;
305
306 for (int row = 0; row < height; ++row) {
307 for (int col = 0; col < width; ++col) {
308
309 Color sum;
310
311 // Stratified supersampling within the pixel.
312 for (int iu = 0; iu < antialiasing_level; iu++) {
313 for (int iv = 0; iv < antialiasing_level; iv++) {
314 float u = (float(iu) + pcg.random_float() ) / antialiasing_level;
315 float v = (float(iv) + pcg.random_float() ) / antialiasing_level;
316 Ray ray = fire_ray(row, col, u, v);
317 sum += func(ray);
318 }
319 }
320
321 // Average all samples collected for this pixel.
322 frame.set_pixel(col, row, sum / float(antialiasing_level * antialiasing_level));
323 }
324 }
