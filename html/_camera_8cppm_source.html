36// Ray struct, used for light rays. Needs to be cache-optimized.
37export struct Ray {
38 Point origin{0.0f, 0.0f, 0.0f};
39 Vec direction{0.0f, 0.0f, 0.0f};
40 float tmin{1e-5f}; // Minimum t value for ray intersection (to avoid self-intersection)
41 float tmax{std::numeric_limits<float>::infinity()};
42 int depth{0}; // Depth of the ray (number of bounces)
43
44 // Methods
45 [[nodiscard]] bool is_close(const Ray& other_ray, float tolerance = 1e-5f) const {
46 return (origin.is_close(other_ray.origin, tolerance) && direction.is_close(other_ray.direction, tolerance));
47 };
48
49 [[nodiscard]] Point at(float t) const {
50 return origin + (direction * t);
51 }

127 return Ray{

129 .direction = trans * direction,

136// FIRE RAY
137// Declared this way, u and v are always between 0 and 1
138Ray OrthogonalCamera::fire_ray(const float u, const float v) const {
139 // Ray origin is on the image plane at distance d from the camera position
140 const Point ray_origin{-1.0f, (1.0f - 2.0f * u) * aspect_ratio, 2.0f * v - 1.0f}; // Camera space origin
141 constexpr Vec ray_direction{1.0f, 0.0f, 0.0f}; // Camera space direction (orthogonal to the image plane)
142 return Ray{ray_origin, ray_direction}.transform(trans);
143}

145Ray PerspectiveCamera::fire_ray(const float u, const float v) const {
146 // Ray origin is the camera position (0,0,0 in camera space)
147 const Point ray_origin{-d, 0.0f, 0.0f}; // Camera space origin
148 const Vec ray_direction{d, (1.0f - 2.0f * u) * aspect_ratio, 2.0f * v - 1.0f}; // Camera space direction (from the camera position to the pixel on the image plane)
149 return Ray{ray_origin, ray_direction}.transform(trans);
150}

164Ray ImageTracer::fire_ray(const int row, const int col, const float u_pixel, const float v_pixel) const {
165 const float width = static_cast<const float>(frame.width);
166 const float height = static_cast<const float>(frame.height);
167
168 const float u = (static_cast<const float>(col) + u_pixel) / width;
169 const float v = 1.0f - (static_cast<const float>(row) + v_pixel) / height;
170
171 return camera.fire_ray(u, v);
172}

176void ImageTracer::fire_all_rays(const std::function<Color(const Ray&)>& func) {
177 const int width = frame.width;
178 const int height = frame.height;
179
180 // Takes a function as an argument: it will be the algorithm to solve the rendering equation
181 for (int row = 0; row < height; ++row) {
182 for (int col = 0; col < width; ++col) {
183 Ray ray = fire_ray(row, col, 0.5f, 0.5f);
184 Color color = func(ray);
185 frame.set_pixel(col, row, color);
186 }
187 }
188}

190void ImageTracer::fire_all_rays(const std::function<Color(const Ray&)>& func, PCG& pcg, int antialiasing_level) {
191 const int width = frame.width;
192 const int height = frame.height;
193 if (antialiasing_level < 1) antialiasing_level = 1;
194
195 // Takes a function as an argument: it will be the algorithm to solve the rendering equation
196 for (int row = 0; row < height; ++row) {
197 for (int col = 0; col < width; ++col) {
198
199 Color sum; // Cumulates sampled Colors
200 for (int iu = 0; iu < antialiasing_level; iu++) {
201 for (int iv = 0; iv < antialiasing_level; iv++) {
202 float u = (float(iu) + pcg.random_float() ) / antialiasing_level;
203 float v = (float(iv) + pcg.random_float() ) / antialiasing_level;
204 Ray ray = fire_ray(row, col, u, v);
205 sum += func(ray);
206 }
207 }
208 frame.set_pixel(col, row, sum / float(antialiasing_level * antialiasing_level));
209 }
210 }
